# Super Mario World 2: Yoshi's Island: playing 1-1, from new_game.txt's
# `level` state: the counts the capture follows, and the real pause screen.
#
#   scripts/check.sh ROMDIR super_mario_world_2_yoshis_island
load level
watch 0x37B 1

# right and jumping (B): the first coins of 1-1 ($037B counts them)
hold RIGHT 20
hold RIGHT+B 25
hold RIGHT 20
hold RIGHT+B 25
until 0x37B 0xFF 0x02 120
expect 0x37B 02 00
expect 0x118 0F
map coins ok                         # 2 coins in the bar

# red coins ($03B4), flowers ($03B8), lives ($0379): the pause screen shows
# what's in them (here 3/20, 2/5, 7)
poke 0x3B4 3
poke 0x3B8 2
poke 0x379 7
wait 1
map counts ok

# paused for real ($0B10 = 01; $0B0F up to 13 as it comes in): no capture,
# the last one stays
untiltap 0xB10 0xFF 0x01 60 START
until 0xB0F 0xFF 0x13 60
expect 0x118 0F
map paused not_live
untiltap 0xB10 0xFF 0x00 120 START
until 0xB0F 0xFF 0x00 60
map unpaused ok
